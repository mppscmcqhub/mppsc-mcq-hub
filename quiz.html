function submitQuiz() { let score = 0; document.querySelectorAll(".qbox").forEach((box, i) => { const q = questions[i]; const selected = box.querySelector(`input[name="q${i}"]:checked`); const result = box.querySelector(".result"); result.style.display = "block"; result.className = "result"; // कुछ भी Select नहीं किया if (selected === null) { result.classList.add("skip"); result.innerHTML = "⏺ अनुत्तरित / Not Answered"; return; } const answer = Number(selected.value); if (answer === q.a) { score++; result.classList.add("correct"); result.innerHTML = `✅ सही / Correct
${q.e}`; } else { result.classList.add("wrong"); result.innerHTML = `❌ गलत / Wrong
${q.e}`; } box.querySelectorAll("input").forEach(r => r.disabled = true); }); document.getElementById("scoreBox").innerHTML = `

🎯 स्कोर / Score: ${score}/${questions.length}

`; document.getElementById("submitBtn").style.display = "none"; document.getElementById("restartBtn").style.display = "inline-block"; }
